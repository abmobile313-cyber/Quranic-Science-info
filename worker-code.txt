const enc = new TextEncoder();
const b64url = buf => btoa(String.fromCharCode(...new Uint8Array(buf))).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
const b64urlStr = s => b64url(enc.encode(s));
const hex = buf => [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, "0")).join("");
const digits = s => String(s || "").replace(/\D/g, "");
const last10 = s => digits(s).slice(-10);
const json = (o, status = 200) => new Response(JSON.stringify(o), { status, headers: { "content-type": "application/json" } });

/* ---------- Razorpay signature ---------- */
export async function verifyRazorpay(body, signature, secret) {
  if (!signature || !secret) return false;
  const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const mac = hex(await crypto.subtle.sign("HMAC", key, enc.encode(body)));
  if (mac.length !== signature.length) return false;
  let d = 0; for (let i = 0; i < mac.length; i++) d |= mac.charCodeAt(i) ^ signature.charCodeAt(i);
  return d === 0;
}

/* ---------- Google service-account token (for Firestore) ---------- */
let TOKEN = { v: "", exp: 0 };
export async function googleToken(sa, fetchImpl = fetch) {
  const now = Math.floor(Date.now() / 1000);
  if (TOKEN.v && TOKEN.exp - 60 > now) return TOKEN.v;
  const header = b64urlStr(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claim = b64urlStr(JSON.stringify({ iss: sa.client_email, scope: "https://www.googleapis.com/auth/datastore https://www.googleapis.com/auth/firebase.messaging", aud: "https://oauth2.googleapis.com/token", iat: now, exp: now + 3600 }));
  const pem = sa.private_key.replace(/-----[^-]+-----/g, "").replace(/\s+/g, "");
  const der = Uint8Array.from(atob(pem), c => c.charCodeAt(0));
  const key = await crypto.subtle.importKey("pkcs8", der, { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" }, false, ["sign"]);
  const sig = b64url(await crypto.subtle.sign("RSASSA-PKCS1-v1_5", key, enc.encode(header + "." + claim)));
  const r = await fetchImpl("https://oauth2.googleapis.com/token", { method: "POST", headers: { "content-type": "application/x-www-form-urlencoded" }, body: "grant_type=urn%3Aietf%3Aparams%3Aoauth%3Agrant-type%3Ajwt-bearer&assertion=" + header + "." + claim + "." + sig });
  const j = await r.json(); if (!j.access_token) throw new Error("google token: " + JSON.stringify(j).slice(0, 200));
  TOKEN = { v: j.access_token, exp: now + (j.expires_in || 3600) };
  return TOKEN.v;
}

/* ---------- Firestore REST (minimal) ---------- */
const toFS = v => v === null || v === undefined ? { nullValue: null }
  : typeof v === "boolean" ? { booleanValue: v }
  : typeof v === "number" ? (Number.isInteger(v) ? { integerValue: String(v) } : { doubleValue: v })
  : typeof v === "string" ? { stringValue: v }
  : Array.isArray(v) ? { arrayValue: { values: v.map(toFS) } }
  : { mapValue: { fields: Object.fromEntries(Object.entries(v).map(([k, x]) => [k, toFS(x)])) } };
const fromFS = f => f == null ? null : "stringValue" in f ? f.stringValue : "integerValue" in f ? Number(f.integerValue) : "doubleValue" in f ? f.doubleValue
  : "booleanValue" in f ? f.booleanValue : "nullValue" in f ? null : "timestampValue" in f ? Date.parse(f.timestampValue)
  : "arrayValue" in f ? (f.arrayValue.values || []).map(fromFS) : "mapValue" in f ? Object.fromEntries(Object.entries(f.mapValue.fields || {}).map(([k, x]) => [k, fromFS(x)])) : null;
const docOut = d => Object.assign({ id: d.name.split("/").pop() }, fromFS({ mapValue: { fields: d.fields || {} } }));

export function makeFS(env, fetchImpl = fetch) {
  const sa = typeof env.FIREBASE_SA === "string" ? JSON.parse(env.FIREBASE_SA) : env.FIREBASE_SA;
  const base = `https://firestore.googleapis.com/v1/projects/${sa.project_id}/databases/(default)/documents`;
  const call = async (url, init = {}) => {
    const t = await googleToken(sa, fetchImpl);
    const r = await fetchImpl(url, Object.assign({}, init, { headers: Object.assign({ authorization: "Bearer " + t, "content-type": "application/json" }, init.headers || {}) }));
    if (r.status === 404) return null;
    const j = await r.json(); if (!r.ok) throw new Error("firestore " + r.status + ": " + JSON.stringify(j).slice(0, 300));
    return j;
  };
  const flatPaths = (o, pre = "") => Object.entries(o).flatMap(([k, v]) => v && typeof v === "object" && !Array.isArray(v) && Object.keys(v).length ? flatPaths(v, pre + "`" + k + "`.") : [pre + "`" + k + "`"]);
  return {
    async get(path) { const d = await call(`${base}/${path}`); return d ? docOut(d) : null; },
    /* merge-write: only the given (nested) fields change */
    async merge(path, data) {
      const mask = flatPaths(data).map(p => "updateMask.fieldPaths=" + encodeURIComponent(p)).join("&");
      return call(`${base}/${path}?${mask}`, { method: "PATCH", body: JSON.stringify({ fields: toFS(data).mapValue.fields }) });
    },
    async add(coll, data) { return call(`${base}/${coll}`, { method: "POST", body: JSON.stringify({ fields: toFS(data).mapValue.fields }) }); },
    async query(coll, where = [], limit = 500) {
      const filters = where.map(([field, op, value]) => ({ fieldFilter: { field: { fieldPath: field }, op, value: toFS(value) } }));
      const structuredQuery = { from: [{ collectionId: coll }], limit };
      if (filters.length === 1) structuredQuery.where = filters[0];
      if (filters.length > 1) structuredQuery.where = { compositeFilter: { op: "AND", filters } };
      const r = await call(`${base}:runQuery`, { method: "POST", body: JSON.stringify({ structuredQuery }) });
      return (r || []).filter(x => x.document).map(x => docOut(x.document));
    }
  };
}

/* ---------- WhatsApp Cloud API ---------- */
export const waNumber = p => { const d = digits(p); return d.length === 10 ? "91" + d : d.replace(/^0+/, ""); };
export async function sendWA(env, to, template, params, fetchImpl = fetch) {
  if (!env.WA_TOKEN || !env.WA_PHONE_ID || !to) return { skipped: true };
  const body = { messaging_product: "whatsapp", to: waNumber(to), type: "template",
    template: { name: template, language: { code: env.WA_LANG || "en" }, components: [{ type: "body", parameters: params.map(t => ({ type: "text", text: String(t).slice(0, 900) })) }] } };
  const r = await fetchImpl(`https://graph.facebook.com/v21.0/${env.WA_PHONE_ID}/messages`, { method: "POST", headers: { authorization: "Bearer " + env.WA_TOKEN, "content-type": "application/json" }, body: JSON.stringify(body) });
  const j = await r.json().catch(() => ({})); return { ok: r.ok, status: r.status, id: (j.messages || [])[0]?.id, error: j.error?.message };
}

/* ---------- free app alert (Firebase Cloud Messaging, no charge) ---------- */
export async function sendPush(env, token, msg, fetchImpl = fetch) {
  const sa = typeof env.FIREBASE_SA === "string" ? JSON.parse(env.FIREBASE_SA) : env.FIREBASE_SA, t = await googleToken(sa, fetchImpl);
  const r = await fetchImpl(`https://fcm.googleapis.com/v1/projects/${sa.project_id}/messages:send`, { method: "POST", headers: { authorization: "Bearer " + t, "content-type": "application/json" },
    body: JSON.stringify({ message: { token, webpush: { headers: { Urgency: "high", TTL: "7200" }, data: { title: msg.title, body: msg.body, url: msg.url || "/", tag: msg.tag || "" } } } }) });
  const j = await r.json().catch(() => ({})), code = JSON.stringify(j.error || "");
  return { ok: r.ok, gone: r.status === 404 || /UNREGISTERED|INVALID_ARGUMENT/.test(code) };
}

export async function sendWAText(env, to, text, fetchImpl = fetch) {
  if (!env.WA_TOKEN || !env.WA_PHONE_ID) return { skipped: true };

  return { ok: r.ok, status: r.status };
}

/* free alert on the Quranic Science Android app (Firebase Cloud Messaging, no charge) */
export async function sendAppPush(env, token, msg, fetchImpl = fetch) {
  const sa = typeof env.FIREBASE_SA === "string" ? JSON.parse(env.FIREBASE_SA) : env.FIREBASE_SA, t = await googleToken(sa, fetchImpl);
  const r = await fetchImpl(`https://fcm.googleapis.com/v1/projects/${sa.project_id}/messages:send`, { method: "POST", headers: { authorization: "Bearer " + t, "content-type": "application/json" },
    body: JSON.stringify({ message: { token, notification: { title: msg.title, body: msg.body }, data: { url: msg.url || "", tag: msg.tag || "" },
      android: { priority: "HIGH", notification: { channel_id: "alerts", icon: "ic_stat", color: "#C99A45", tag: msg.tag || undefined } } } }) });
  const j = await r.json().catch(() => ({})), code = JSON.stringify(j.error || "");
  return { ok: r.ok, gone: r.status === 404 || /UNREGISTERED|INVALID_ARGUMENT/.test(code) };
}
export async function notify(env, fs, user, msg, now = Date.now(), fetchImpl = fetch) {
  if (user.pushApp && user.pushApp.token) {
    const r = await sendAppPush(env, user.pushApp.token, msg, fetchImpl).catch(() => ({ ok: false }));
    if (r.ok) { if (user.push && user.push.token) await sendPush(env, user.push.token, msg, fetchImpl).catch(() => {}); return { via: "app" }; }
    if (r.gone && user.id) await fs.merge(`users/${user.id}`, { pushApp: { token: "" } });
  }
  if (user.push && user.push.token) {
    const r = await sendPush(env, user.push.token, msg, fetchImpl).catch(() => ({ ok: false }));
    if (r.ok) return { via: "push" };
    if (r.gone && user.id) await fs.merge(`users/${user.id}`, { push: { token: "" } });
  }
  const ph = waNumber(user.waPhone || user.phone); if (!ph || !env.WA_TOKEN) return { via: "none" };
  const inbox = await fs.get(`waInbox/${ph}`);
  if (inbox && now - inbox.ts < 23 * 3600000) { const r = await sendWAText(env, ph, `${msg.title}\n${msg.body}${msg.url ? "\n" + msg.url : ""}`, fetchImpl); if (r.ok) return { via: "whatsapp-free" }; }
  if (!msg.paid) return { via: "none" };
  const mk = "m" + new Date(now + 330 * 60000).toISOString().slice(0, 7).replace("-", "_"), cnt = (await fs.get("config/waCount")) || {}, used = cnt[mk] || 0, limit = +(env.WA_MONTHLY_LIMIT || 200);
  if (used >= limit) return { via: "limit-reached" };
  const r = await sendWA(env, ph, msg.template, msg.params, fetchImpl);
  if (r.ok) { await fs.merge("config/waCount", { [mk]: used + 1 }); return { via: "whatsapp-paid" }; }
  return { via: "failed", error: r.error || r.status };
}

/* translate a bot reply line by line; lines with links stay as they are (free Google translate, original if it fails) */
export async function translateLines(text, tl, fetchImpl = fetch) {
  const lines = String(text).split("\n"); const out = [];
  for (const ln of lines) {
    if (!ln.trim() || /https?:\/\//.test(ln) || /[\u0600-\u06ff\u0900-\u097f]/.test(ln) && !/[a-z]{3}/i.test(ln)) { out.push(ln); continue; }
    try { const r = await fetchImpl(`https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=${tl}&dt=t&q=${encodeURIComponent(ln.replace(/\*/g, ""))}`); const j = await r.json(); const o = (j[0] || []).map(x => x[0]).join(""); out.push(o.trim() ? o : ln); } catch (e) { out.push(ln); }
  }
  return out.join("\n");
}
/* the teacher's answer reaches the student in the language the student asked in (free Google translate; original if it fails) */
const scriptOf = t => /[\u0600-\u06ff]/.test(t) ? (/[ٹڈڑںےۓ]/.test(t) ? "ur" : "ar") : /[\u0900-\u097f]/.test(t) ? "hi" : "en";
export async function toLangOf(text, like, fetchImpl = fetch) {
  text = String(text || ""); const tl = scriptOf(String(like || "")); if (!text.trim() || scriptOf(text) === tl) return text;
  try { const r = await fetchImpl(`https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=${tl}&dt=t&q=${encodeURIComponent(text.slice(0, 1500))}`); const j = await r.json(); const o = (j[0] || []).map(x => x[0]).join(""); return o.trim() ? o : text; } catch (e) { return text; }
}

/* ---------- weekly star students: attendance + 3-question answers + teacher's stars, per course and batch ---------- */
const slugW = s => String(s || "").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
export function leaderRows(classes, users, revs, now, ck, b) {
  const from = now - 7 * 864e5, to = now - 3600e3;
  const held = classes.filter(k => k.course === ck && (!k.batch || k.batch === "ALL" || k.batch === b)).filter(k => { const t = istMs(k.date); return !isNaN(t) && t >= from && t <= to; });
  const rows = [];
  for (const u of users) {
    if (!(u.access || {})[ck] || (u.batch || "") !== b) continue;
    const A = ((u.att || {})[ck]) || {}, Q = (((u.prog || {})[ck]) || {}).qz || {};
    const att = held.filter(k => { const a = A[slugW(k.id + "-" + k.id)]; return a && a.p !== false; }).length;
    const quiz = held.filter(k => Q[k.id] && Q[k.id].ok).length;
    const r = revs.filter(x => x.uid === u.id && x.course === ck && (x.ts || 0) > now - 14 * 864e5).sort((x, y) => (y.ts || 0) - (x.ts || 0))[0];
    const stars = r ? +r.stars || 0 : 0;
    const score = Math.round((held.length ? att / held.length * 40 + quiz / held.length * 30 : 0) + (stars ? stars / 5 * 30 : 0));
    if (score > 0) rows.push({ name: String(u.name || "Student").slice(0, 40), score, att, quiz, held: held.length, stars });
  }
  return rows.sort((x, y) => y.score - x.score || y.stars - x.stars || x.name.localeCompare(y.name)).slice(0, 5);
}

/* Makes sure our WhatsApp Business Account sends incoming messages to this app (subscribed_apps). Safe to run many times. */
export async function waSubscribe(env, extra = "", fetchImpl = fetch) {
  if (!env.WA_TOKEN) return { skipped: true };
  const G = "https://graph.facebook.com/v21.0", H = { authorization: "Bearer " + env.WA_TOKEN };

  out.debug = null;

    out.debug = d.error ? d.error.message : ((d.data || {}).granular_scopes || []).map(g => g.scope + ":" + (g.target_ids || []).length).join(" ");

  } catch (e) {}
  for (const id of ids) {
    const row = { id };
    try { const r = await (await fetchImpl(`${G}/${id}/subscribed_apps`, { method: "POST", headers: H })).json(); row.subscribe = r.error ? { error: r.error.message } : r; } catch (e) { row.subscribe = { error: e.message }; }

  }

export function cleanCustomer(c = {}) {
  const a = c.address || {};
  return { name: clip(c.name, 80), phone: clip(String(c.phone || "").replace(/[^\d+]/g, ""), 16), email: clip(c.email, 120).toLowerCase(),
    address: { line1: clip(a.line1, 160), line2: clip(a.line2, 160), city: clip(a.city, 60), state: clip(a.state, 60), pin: clip(a.pin, 12), country: clip(a.country || "India", 40) } };

export async function shopOrder(env, fs, body, now = Date.now()) {
  const brand = String(body.brand || ""); if (!SHOP_BRANDS[brand]) return { error: "unknown shop" };
  const cust = cleanCustomer(body.customer); if (!cust.name || cust.phone.replace(/\D/g, "").length < 10) return { error: "Please give your name and a valid mobile number." };
  const cfg = ((await fs.get("config/shop")) || {})[brand] || {};
  let lines = [], quoteStage = "";
  if (body.quoteId) {
    const q = await fs.get(`quotes/${body.quoteId}`);
    const split = SHOP_SPLIT[brand], stage = q && q.status === "advance_paid" ? "final" : "full";
    if (!q || q.key !== body.quoteKey || q.brand !== brand || !(q.status === "quoted" || (split && q.status === "advance_paid")) || !(+q.price > 0)) return { error: "This quote is not ready for payment yet." };
    let amt = +q.price, label = "";
    if (split && stage === "full") { amt = Math.round(q.price * split); label = ` – ${Math.round(split * 100)}% advance`; quoteStage = "advance"; }
    else if (split) { amt = Math.max(0, Math.round(q.price - (+q.advancePaid || Math.round(q.price * split)))); label = ` – ${100 - Math.round(split * 100)}% final payment`; quoteStage = "final"; }
    lines = [{ id: "quote:" + q.id, name: `${q.service || "Project"} (quote ${q.id.slice(0, 6)})${label}`, price: amt, qty: 1 }];
  } else {
    const want = (Array.isArray(body.items) ? body.items : []).slice(0, 40);
    for (const w of want) {
      const it = await fs.get(`shopItems/${String(w.id || "").replace(/[^\w-]/g, "")}`);
      const qty = Math.max(1, Math.min(50, Math.floor(+w.qty || 1)));
      if (!it || it.brand !== brand || it.active === false || !(+it.price > 0)) return { error: `"${clip(w.name || w.id, 60)}" is not available right now.` };
      lines.push({ id: it.id, name: it.name + (w.variant ? " – " + clip(w.variant, 40) : ""), price: +it.price, qty });
    }
  }
  if (!lines.length) return { error: "Your cart is empty." };
  const sub = lines.reduce((t, l) => t + l.price * l.qty, 0), ship = sub >= (+cfg.freeShipAbove || Infinity) ? 0 : (+cfg.ship || 0), total = Math.round((sub + ship) * 100) / 100;
  const id = (brand.slice(0, 2).toUpperCase()) + new Date(now + 330 * 60000).toISOString().slice(2, 10).replace(/-/g, "") + "-" + rid(5).toUpperCase(), key = rid(16);
  await fs.merge(`orders/${id}`, { brand, key, items: lines, subtotal: sub, ship, total, currency: shopCur(brand), customer: cust, note: clip(body.note, 500), quoteId: body.quoteId || "", quoteStage, status: "awaiting_payment", timeline: [{ s: "awaiting_payment", ts: now }], ts: now });
  return { id, key, total, currency: shopCur(brand), rzpKey: shopKey(env, brand), brandName: SHOP_BRANDS[brand], customer: cust };
}
